def abd_matrix(angles: list[float], thickness: float) -> np.ndarray:
    """Return the 6x6 ABD matrix. Ply 1 is the top ply."""
    n_plies = len(angles)
    z = thickness * (np.arange(n_plies + 1) - n_plies / 2.0)
